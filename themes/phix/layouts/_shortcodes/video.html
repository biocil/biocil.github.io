{{- /* Usage: {{< video src="clip.mp4" poster="clip.jpg" controls="yes" caption="…" >}} The file is looked up in the page folder, then in assets/media/, then used as a URL. Without `controls`, the video plays muted and loops (paused for visitors who prefer reduced motion). */ -}} {{- $src := .Get "src" -}} {{- $url := $src -}}{{- $type := strings.TrimPrefix "." (path.Ext $src) -}}{{- $poster := "" -}} {{- $posterName := .Get "poster" | default (replace $src (path.Ext $src) ".jpg") -}} {{- if not (strings.HasPrefix $src "http") -}} {{- with or (.Page.Resources.GetMatch $src) (resources.Get (path.Join "media" $src)) -}}{{- $url = .RelPermalink -}}{{- $type = .MediaType.SubType -}}{{- end -}} {{- with or (.Page.Resources.GetMatch $posterName) (resources.Get (path.Join "media" $posterName)) -}}{{- $poster = .RelPermalink -}}{{- end -}} {{- end -}}
